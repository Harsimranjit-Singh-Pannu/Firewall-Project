from actions import (
    ContinueAction,
    JumpToRuleset

from rulesets.argument_binding import ArgumentBinding

from conditions.enums import (
    SourceType,
    RightSourceType
)


class ActionExecutor:

    def build(
        self,
        action_config
    ):

        if action_config is None:

            return None

        action_type = action_config.type

        parameters = (
            action_config.parameters
            or {}
        )

        if action_type == "CONTINUE":

            return ContinueAction()

        if action_type == "JUMP":

            ruleset_name = parameters.get(
                "ruleset_name"
            )

            if not ruleset_name:

                raise ValueError(
                    "JUMP action requires 'ruleset_name'"
                )

            argument_bindings = self.build_argument_bindings(
                parameters.get(
                    "argument_bindings",
                    []
                )
            )

            return JumpToRuleset(

                ruleset_name=ruleset_name,

                argument_bindings=argument_bindings

            )

        raise ValueError(
            f"Unsupported action type: {action_type}"
        )

    def build_argument_bindings(
        self,
        bindings
    ):

        result = []

        for binding in bindings:

            if isinstance(
                binding,
                ArgumentBinding
            ):

                binding.validate()

                result.append(
                    binding
                )

                continue

            if not isinstance(
                binding,
                dict
            ):

                raise ValueError(
                    "Argument binding must be a dictionary"
                )

            source_type = binding.get(
                "source_type"
            )

            if isinstance(
                source_type,
                str
            ):

                if source_type == "Parameter":

                    source_type = SourceType.PARAMETER

                elif source_type == "User_Value":

                    source_type = RightSourceType.USER_VALUE

                else:

                    raise ValueError(
                        f"Unsupported argument source type: {source_type}"
                    )

            argument_binding = ArgumentBinding(

                argument_name=binding[
                    "argument_name"
                ],

                source_type=source_type,

                source_name=binding.get(
                    "source_name"
                ),

                value=binding.get(
                    "value"
                )

            )

            argument_binding.validate()

            result.append(
                argument_binding
            )

        return result

from .resolver import ConditionResolver

from .comparators import ComparatorFactory

from .enums import (
    SourceType,
    Operator
)



class ConditionEvaluator:


    def __init__(

        self

    ):

        self.resolver = ConditionResolver()



    def evaluate(

        self,

        condition,

        context,

        arguments=None

    ):


        left = self.resolver.resolve_left(

            condition,

            context,

            arguments

        )


        right = self.resolver.resolve_right(

            condition,

            context,

            arguments

        )


        #
        # Exists / Does Not Exist
        #

        if condition.operator == Operator.EXISTS:

            return left is not None



        if condition.operator == Operator.DOES_NOT_EXIST:

            return left is None



        if left is None:

            return False



        #
        # Handle scalar and collections
        #

        values = (

            left

            if isinstance(left, list)

            else [left]

        )



        #
        # Determine datatype
        #

        datatype = "String"



        if condition.left_type == SourceType.PARAMETER:


            parameter = context.get_parameter(

                condition.left_name

            )


            if parameter is not None:

                datatype = parameter.datatype



        elif condition.left_type == SourceType.ARGUMENT:


            if arguments is not None:


                value = arguments.get(

                    condition.left_name

                )


                if isinstance(value, bool):

                    datatype = "Boolean"


                elif isinstance(value, int):

                    datatype = "Integer"


                elif isinstance(value, str):

                    datatype = "String"



        #
        # Comparator
        #

        comparator = ComparatorFactory.get(

            datatype

        )



        results = []


        for value in values:


            results.append(

                comparator.compare(

                    value,

                    right,

                    condition.operator

                )

            )



        #
        # Match mode
        #

        if condition.match_mode.value == "Any":

            return any(results)



        if condition.match_mode.value == "All":

            return all(results)



        if condition.match_mode.value == "None":

            return not any(results)



        return False

class ConditionResolver:


    def resolve_left(
        self,
        condition,
        context,
        arguments=None
    ):

        if condition.left_type == SourceType.PARAMETER:

            parameter = context.get_parameter(
                condition.left_name
            )


            if parameter is not None:

                return parameter.value



        if condition.left_type == SourceType.ARGUMENT:

            if arguments:

                return arguments.get(
                    condition.left_name
                )


        return None



    def resolve_right(
        self,
        condition,
        context,
        arguments=None
    ):


        if condition.right_type == RightSourceType.PARAMETER:

            parameter = context.get_parameter(
                condition.right_value
            )


            if parameter is not None:

                return parameter.value



        if condition.right_type == RightSourceType.ARGUMENT:

            if arguments:

                return arguments.get(
                    condition.right_value
                )


        return condition.right_value

class BooleanComparator(Comparator):


    def to_bool(

        self,

        value

    ):


        if isinstance(

            value,

            bool

        ):

            return value


        return str(value).lower() in (

            "true",

            "1",

            "yes"

        )



    def compare(

        self,

        left,

        right,

        operator

    ):


        left = self.to_bool(left)

        right = self.to_bool(right)


        if operator == Operator.EQUALS:

            return left == right


        if operator == Operator.NOT_EQUALS:

            return left != right


        raise ValueError(
            f"Unsupported operator {operator}"
        )

        datatype = str(
            datatype

        if datatype == "integer":

            return IntegerComparator()

        if datatype == "boolean":

            return BooleanComparator()

        if datatype == "ipv4":

            return IPv4Comparator()

        return StringComparator()

from conditions.enums import Operator

from .base import Comparator



class StringComparator(Comparator):


    def compare(

        self,

        left,

        right,

        operator

    ):


        left = str(left)

        right = str(right)


        if operator == Operator.EQUALS:

            return left == right


        if operator == Operator.NOT_EQUALS:

            return left != right


        if operator == Operator.CONTAINS:

            return right in left


        if operator == Operator.NOT_CONTAINS:

            return right not in left


        if operator == Operator.STARTS_WITH:

            return left.startswith(right)


        if operator == Operator.ENDS_WITH:

            return left.endswith(right)


        if operator == Operator.REGEX_MATCH:

            return re.search(right, left) is not None


        if operator == Operator.REGEX_NOT_MATCH:

            return re.search(right, left) is None


        raise ValueError(

            f"Unsupported operator {operator}"

        )

from configuration.models.action_config import ActionConfig

    SourceType,
    RightSourceType
)


class ActionSerializer:

    def serialize(
        self,
        action: ActionConfig
    ):

        if action is None:

            return None

        if not isinstance(
            action,
            ActionConfig
        ):

            raise ValueError(
                f"Unsupported action type: {type(action)}"
            )

        parameters = dict(
            action.parameters
        )

        argument_bindings = parameters.get(
            "argument_bindings"
        )

        if argument_bindings is not None:

            parameters["argument_bindings"] = [

                self.serialize_argument_binding(
                    binding
                )

                for binding in argument_bindings

            ]

        return {

            "id": str(action.id),

            "type": action.type,

            "parameters": parameters

        }

    def serialize_argument_binding(
        self,
        binding
    ):

        if not isinstance(
            binding,
            ArgumentBinding
        ):

            raise ValueError(
                f"Unsupported argument binding type: "
                f"{type(binding)}"
            )

        return {

            "argument_name": binding.argument_name,

            "source_type": binding.source_type.value,

            "source_name": binding.source_name,

            "value": binding.value

        }

    def deserialize(
        self,
        data
    ):

        if data is None:

            return None

        action_id = data.get(
            "id"
        )

        parameters = dict(
            data.get(
                "parameters",
                {}
            )
        )

        argument_bindings = parameters.get(
            "argument_bindings"
        )

        if argument_bindings is not None:

            parameters["argument_bindings"] = [

                self.deserialize_argument_binding(
                    binding
                )

                for binding in argument_bindings

            ]

        return ActionConfig(

            type=data["type"],

            parameters=parameters,

            id=(

                UUID(action_id)

                if action_id

                else uuid4()

            )

        )

    def deserialize_argument_binding(
        self,
        data
    ):

        source_type = data.get(
            "source_type"
        )

        if source_type == SourceType.PARAMETER.value:

            resolved_source_type = SourceType.PARAMETER

        elif source_type == RightSourceType.USER_VALUE.value:

            resolved_source_type = RightSourceType.USER_VALUE

        else:

            raise ValueError(
                f"Unsupported argument binding source: "
                f"{source_type}"
            )

        return ArgumentBinding(

            argument_name=data["argument_name"],

            source_type=resolved_source_type,

            source_name=data.get(
                "source_name"
            ),

            value=data.get(
                "value"
            )

class ArithmeticOperator(Enum):

    ADD = "Add"
    SUBTRACT = "Subtract"
    MULTIPLY = "Multiply"
    DIVIDE = "Divide"



def execute(
    value1,
    value2,
    operator
):

    if operator == ArithmeticOperator.ADD:

        result = value1 + value2


    elif operator == ArithmeticOperator.SUBTRACT:

        result = value1 - value2


    elif operator == ArithmeticOperator.MULTIPLY:

        result = value1 * value2


    elif operator == ArithmeticOperator.DIVIDE:

        if value2 == 0:

            raise ValueError(
                "Division by zero is not allowed"
            )

        result = value1 / value2


    else:

        raise ValueError(
            f"Unsupported operator: {operator}"
        )


    return {

        "success": True,

        "operation": "Arithmetic",

        "result": result

    }

class SetParameterOperation:

    name = "set-parameter"

    def execute(
        self,
        context,
        parameters
    ):

        if not isinstance(
            parameters,
            dict
        ):

            raise ValueError(
                "set-parameter parameters must be a dictionary"
            )

        if "name" not in parameters:

            raise ValueError(
                "set-parameter requires 'name'"
            )

        if "value" not in parameters:

            raise ValueError(
                "set-parameter requires 'value'"
            )

        name = parameters["name"]

        value = parameters["value"]

        datatype = parameters.get(
            "datatype",
            "String"
        )

        scope = parameters.get(
            "scope",
            ParameterScope.LOCAL
        )

        if isinstance(
            scope,
            str
        ):

            try:

                scope = ParameterScope(
                    scope
                )

            except ValueError:

                raise ValueError(
                    f"Invalid parameter scope: {scope}"
                )

        if not isinstance(
            scope,
            ParameterScope
        ):

            raise ValueError(
                "scope must be a ParameterScope or valid scope string"
            )

        context.set_parameter(

            name=name,

            value=value,

            datatype=datatype,

            scope=scope

        )

        return True

                operation,

                context

            )



        return rule.action.execute(
            context
        )

from actions import (
    ContinueAction,
    JumpToRuleset
)

from conditions.enums import (
    SourceType,
    RightSourceType
)



class RulesetExecutor:


    MAX_JUMP_DEPTH = 50



    def __init__(
        self,
        rule_evaluator,
        registry
    ):

        self.rule_evaluator = rule_evaluator

        self.registry = registry



    def resolve_arguments(
        self,
        bindings,
        context
    ):

        if not bindings:

            return None


        arguments = {}


        for binding in bindings:


            binding.validate()



            if binding.source_type == SourceType.PARAMETER:


                parameter = context.get_parameter(

                    binding.source_name

                )


                if parameter is None:

                    raise ValueError(

                        f"Missing parameter: {binding.source_name}"

                    )


                arguments[

                    binding.argument_name

                ] = parameter.value



            elif binding.source_type == RightSourceType.USER_VALUE:


                arguments[

                    binding.argument_name

                ] = binding.value



            else:

                raise ValueError(

                    f"Unsupported argument source: {binding.source_type}"

                )


        return arguments



    def execute(
        self,
        ruleset,
        context,
        arguments=None,
        depth=0
    ):


        if depth > self.MAX_JUMP_DEPTH:

            raise RuntimeError(

                "Maximum ruleset jump depth exceeded"

            )



        ruleset.validate()


        arguments = ruleset.validate_arguments(

            arguments

        )



        context.enter_ruleset(

            ruleset.name

        )


        try:


            for rule in ruleset.rules:


                result = self.rule_evaluator.evaluate(

                    rule,

                    context,

                    arguments

                )



                if result is None:

                    continue



                if isinstance(

                    result,

                    ContinueAction

                ):

                    continue



                if isinstance(

                    result,

                    JumpToRuleset

                ):


                    target = self.registry.get(

                        result.ruleset_name

                    )


                    if target is None:

                        raise ValueError(

                            f"Ruleset not found: {result.ruleset_name}"

                        )



                    target_arguments = self.resolve_arguments(

                        result.argument_bindings,

                        context

                    )



                    self.execute(

                        target,

                        context,

                        target_arguments,

                        depth + 1

                    )


                    continue



                return result



        finally:

            context.exit_ruleset()



        return None

from .argument import Argument



class Ruleset:


    def __init__(
        self,
        name,
        description=None
    ):

        self.name = name

        self.description = description

        self.arguments = []

        self.rules = []



    def add_argument(
        self,
        argument: Argument
    ):

        if self.get_argument(argument.name):

            raise ValueError(
                f"Argument already exists: {argument.name}"
            )


        self.arguments.append(argument)



    def add_rule(
        self,
        rule
    ):

        self.rules.append(rule)



    def get_argument(
        self,
        name
    ):

        for argument in self.arguments:

            if argument.name == name:

                return argument


        return None



    def get_rule(
        self,
        name
    ):

        for rule in self.rules:

            if rule.name == name:

                return rule


        return None



    def validate_arguments(
        self,
        arguments=None
    ):

        arguments = arguments or {}


        #
        # Ruleset accepts no arguments
        #

        if not self.arguments:


            if arguments:

                raise ValueError(

                    f"Ruleset {self.name} does not accept arguments"

                )


            return {}



        #
        # Check unknown arguments first
        #

        for name in arguments:


            if self.get_argument(name) is None:

                raise ValueError(

                    f"Unknown argument: {name}"

                )



        #
        # Resolve arguments and defaults
        #

        resolved = {}


        for argument in self.arguments:


            if argument.name in arguments:

                resolved[

                    argument.name

                ] = arguments[argument.name]


            elif argument.default_value is not None:

                resolved[

                    argument.name

                ] = argument.default_value


            else:

                raise ValueError(

                    f"Missing argument: {argument.name}"

                )



        #
        # Validate resolved arguments
        #

        self.validate_argument_types(

            resolved

        )


        return resolved



    def validate_argument_types(
        self,
        arguments
    ):

        if not arguments:

            return True



        for name, value in arguments.items():


            argument = self.get_argument(

                name

            )


            if argument is None:

                raise ValueError(

                    f"Unknown argument: {name}"

                )



            datatype = argument.datatype.value

            collection_type = argument.collection_type.value



            if collection_type in (

                "Set",

                "Node"

            ):


                if not isinstance(

                    value,

                    (list, set, tuple)

                ):

                    raise ValueError(

                        f"Argument {name} expects collection value"

                    )


                values = value



            else:

                values = [

                    value

                ]



            for item in values:


                if datatype == "String":

                    valid = isinstance(

                        item,

                        str

                    )


                elif datatype == "Integer":

                    valid = isinstance(

                        item,

                        int

                    ) and not isinstance(

                        item,

                        bool

                    )


                elif datatype == "Boolean":

                    valid = isinstance(

                        item,

                        bool

                    )


                elif datatype in (

                    "IPv4",

                    "IPv6"

                ):

                    valid = isinstance(

                        item,

                        str

                    )


                else:

                    valid = True



                if not valid:

                    raise ValueError(

                        f"Argument {name} expects {datatype}"

                    )


        return True



    def validate(self):

        if not self.rules:

            raise ValueError(

                "Ruleset requires at least one rule"

            )


        return True

class RuntimeBuilder:

    def __init__(
        self
    ):

        self.action_executor = ActionExecutor()

    def build_ruleset(
        self,
        ruleset_config: RulesetConfig
    ):

        ruleset = Ruleset(

            name=ruleset_config.name

        )

        for rule_config in ruleset_config.rules:

            rule = self.build_rule(

                rule_config

            )

            ruleset.add_rule(

                rule

            )

        return ruleset

    def build_rulesets(
        self,
        firewall_configuration
    ):

        runtime_rulesets = []

        for ruleset_config in firewall_configuration.rulesets:

            runtime_rulesets.append(

                self.build_ruleset(

                    ruleset_config

                )

            )

        return runtime_rulesets

    def build_rule(
        self,
        rule_config
    ):

        rule = Rule(

            name=rule_config.name

        )

        #
        # Conditions
        #

        condition_group = self.build_condition_group(

            rule_config.condition_group

        )

        rule.set_condition_group(

            condition_group

        )

        #
        # Operations
        #

        for operation in rule_config.operations:

            rule.add_operation(

                operation

            )

        #
        # Action
        #

        runtime_action = self.action_executor.build(

            rule_config.action

        )

        rule.set_action(

            runtime_action

        )

        return rule

    def build_condition_group(
        self,
        group_config
    ):

        if group_config is None:

            return None

        try:

            group_operator = GroupOperator(
                group_config.operator
            )

        except ValueError:

            raise ValueError(
                f"Invalid condition group operator: "
                f"{group_config.operator}"
            )

        conditions = []

        for condition_config in group_config.conditions:

            conditions.append(

                self.build_condition(

                    condition_config

                )

            )

        return ConditionGroup(

            operator=group_operator,

            conditions=conditions

        )

    def build_condition(
        self,
        condition_config
    ):

        condition = copy.deepcopy(
            condition_config
        )

        try:

            condition.left_type = SourceType(
                condition.left_type
            )

        except ValueError:

            raise ValueError(
                f"Invalid condition left_type: "
                f"{condition.left_type}"
            )

        try:

            condition.operator = Operator(
                condition.operator
            )

        except ValueError:

            raise ValueError(
                f"Invalid condition operator: "
                f"{condition.operator}"
            )

        try:

            condition.match_mode = MatchMode(
                condition.match_mode
            )

        except ValueError:

            raise ValueError(
                f"Invalid condition match_mode: "
                f"{condition.match_mode}"
            )

        if condition.right_type is not None:

            try:

                condition.right_type = RightSourceType(
                    condition.right_type
                )

            except ValueError:

                raise ValueError(
                    f"Invalid condition right_type: "
                    f"{condition.right_type}"
                )

        return condition

                file
            )

        file_path = self.folder / f"{name}.json"

from uuid import UUID


class VersionRepository:

    def __init__(
        self
    ):

        self.versions = []

    def save(
        self,
        version
    ):

        existing = self.get(
            version.id
        )

        if existing is not None:

            index = self.versions.index(
                existing
            )

            self.versions[index] = version

        else:

            self.versions.append(
                version
            )

        return version

    def list(
        self
    ):

        return list(
            self.versions
        )

    def get(
        self,
        version_id
    ):

        if isinstance(
            version_id,
            str
        ):

            try:

                version_id = UUID(
                    version_id
                )

            except ValueError:

                return None

        for version in self.versions:

            if version.id == version_id:

                return version

        return None

    def clear(
        self
    ):

        self.versions.clear()

import copy

from uuid import UUID
from datetime import datetime, timezone

from versions.version import Version

from configuration.serializers.firewall_configuration_serializer import (
    FirewallConfigurationSerializer
)

from configuration.serializers.firewall_configuration_deserializer import (
    FirewallConfigurationDeserializer
)

from storage.json_storage import JsonStorage


class VersionService:

    def __init__(
        self,
        repository,
        serializer=None,
        deserializer=None,
        storage=None
    ):

        self.repository = repository

        self.serializer = (
            serializer
            or FirewallConfigurationSerializer()
        )

        self.deserializer = (
            deserializer
            or FirewallConfigurationDeserializer()
        )

        self.storage = (
            storage
            or JsonStorage()
        )

        self._load_versions()

    def _load_versions(
        self
    ):

        self.repository.clear()

        number = 1

        while True:

            try:

                data = self.storage.load(
                    f"version_{number}"
                )

            except ValueError:

                break

            configuration_data = data.get(
                "configuration",
                {
                    "rulesets": []
                }
            )

            configuration = self.deserializer.deserialize(
                configuration_data
            )

            version_id = data.get(
                "id"
            )

            if version_id:

                version_uuid = UUID(
                    version_id
                )

            else:

                version_uuid = None

            created_at_value = data.get(
                "created_at"
            )

            if created_at_value:

                created_at = datetime.fromisoformat(
                    created_at_value
                )

            else:

                created_at = datetime.now(
                    timezone.utc
                )

            version = Version(

                number=data.get(
                    "version",
                    number
                ),

                configuration=configuration,

                **(
                    {
                        "id": version_uuid
                    }
                    if version_uuid
                    else {}
                ),

                created_at=created_at,

                status=data.get(
                    "status",
                    "AVAILABLE"
                )

            )

            self.repository.save(
                version
            )

            number += 1

    def create_version(
        self,
        configuration
    ):

        versions = self.repository.list()

        number = (

            max(
                (
                    version.number
                    for version in versions
                ),
                default=0
            )

            + 1

        )

        configuration_copy = copy.deepcopy(
            configuration
        )

        version = Version(

            number=number,

            configuration=configuration_copy

        )

        self._save_version(
            version
        )

        return version

    def list_versions(
        self
    ):

        return self.repository.list()

    def get_version(
        self,
        version_id
    ):

        return self.repository.get(
            version_id
        )

    def switch_version(
        self,
        version_id
    ):

        version = self.repository.get(
            version_id
        )

        if version is None:

            raise ValueError(
                "Version not found"
            )

        versions = self.repository.list()

        for item in versions:

            item.status = "AVAILABLE"

            self._save_version(
                item
            )

        version.status = "ACTIVE"

        self._save_version(
            version
        )

        return version

    def push_version(
        self,
        version_id
    ):

        version = self.repository.get(
            version_id
        )

        if version is None:

            raise ValueError(
                "Version not found"
            )

        version.status = "ACTIVE"

        self._save_version(
            version
        )

        return version

    def export_version(
        self,
        version_id
    ):

        version = self.repository.get(
            version_id
        )

        if version is None:

            raise ValueError(
                "Version not found"
            )

        return self.serializer.export(
            version.configuration
        )

    def _save_version(
        self,
        version
    ):

        self.storage.save(

            f"version_{version.number}",

            {

                "id": str(
                    version.id
                ),

                "version": version.number,

                "created_at": version.created_at.isoformat(),

                "status": version.status,

                "configuration": self.serializer.serialize(
                    version.configuration
                )

            }

        )

        self.repository.save(
            version
        )
