import json

from configuration.models.firewall_config import FirewallConfiguration

from drafts.draft_service import DraftService

def test_operation_survives_version_commit_and_reload(
    tmp_path
):

    version_repository = VersionRepository()

    version_service = VersionService(
        repository=version_repository
    )

    configuration = FirewallConfiguration()

    base_version = version_service.create_version(
        configuration=configuration,
        name="Base Version"
    )

    draft_repository = DraftRepository()

    draft_service = DraftService(
        repository=draft_repository,
        version_service=version_service
    )

    draft = draft_service.create_draft(

        version_id=base_version.id,

        configuration=base_version.configuration

    )

    ruleset_service = RulesetService(
        draft_service
    )

    ruleset = ruleset_service.create_ruleset(
        draft.id,
        "Web Security"
    )

    rule = ruleset_service.add_rule(
        draft.id,
        ruleset.id,
        "Allow Web Traffic"
    )

    rule_service = RuleService(
        draft_service
    )

    operation = rule_service.create_operation(

        draft.id,

        ruleset.id,

        rule.id,

        "set-parameter",

        {
            "name": "risk",
            "value": 10,
            "scope": "REQUEST"
        }

    )

    committed_version = version_service.create_version(

        configuration=draft.configuration,

        name="Web Traffic Version"

    )

    assert (
        len(
            committed_version.configuration.rulesets
        )
        == 1
    )

    committed_ruleset = (
        committed_version.configuration.rulesets[0]
    )

    assert len(
        committed_ruleset.rules
    ) == 1

    committed_rule = (
        committed_ruleset.rules[0]
    )

    assert len(
        committed_rule.operations
    ) == 1

    committed_operation = (
        committed_rule.operations[0]
    )

    assert (
        committed_operation.name
        == "set-parameter"
    )

    assert (
        committed_operation.id
        == operation.id
    )

    assert (
        committed_operation.parameters
        == {
            "name": "risk",
            "value": 10,
            "scope": "REQUEST"
        }
    )

class RuntimeRuleset:

    """
    Small runtime ruleset implementation matching what
    RulesetExecutor expects.
    """

    def __init__(
        self,
        name,
        rules,
        arguments=None,
    ):
        self.name = name
        self.rules = rules
        self.arguments = arguments or {}

    def validate(self):
        for rule in self.rules:
            rule.validate()

        return True

    def validate_arguments(self, arguments):
        arguments = arguments or {}

        for name, definition in self.arguments.items():

            if name not in arguments:

                if definition.default_value is not None:
                    arguments[name] = definition.default_value
                    continue

                raise ValueError(
                    f"Missing argument: {name}"
                )

        return arguments


def make_condition_group(*conditions, operator=GroupOperator.AND):

    return ConditionGroup(
        operator,
        list(conditions),
    )


def make_runtime_rule(
    condition_group,
    action=None,
    operations=None,
    name="rule",
):

    rule = Rule(name)

    rule.set_condition_group(
        condition_group
    )

    for operation in operations or []:
        rule.add_operation(operation)

    rule.set_action(
        action or ContinueAction()
    )

    return rule

class TestParameterStore:

    def test_store_and_retrieve_parameter(self):

        store = ParameterStore()

        parameter = Parameter(
            name="x",
            value=10,
            datatype="Integer",
        )

        store.set(parameter)

        assert store.get("x") == 10
        assert store.get_parameter("x") == parameter
        assert store.exists("x")

    def test_remove_parameter(self):

        store = ParameterStore()

        store.set(
            Parameter(
                name="x",
                value=10,
                datatype="Integer",
            )
        )

        store.remove("x")

        assert not store.exists("x")
        assert store.get("x") is None

    def test_higher_scope_cannot_be_overwritten_by_lower_scope(self):

        store = ParameterStore()

        store.set(
            Parameter(
                name="x",
                value="system",
                datatype="String",
                scope=ParameterScope.SYSTEM,
            )
        )

        with pytest.raises(ValueError):
            store.set(
                Parameter(
                    name="x",
                    value="local",
                    datatype="String",
                    scope=ParameterScope.LOCAL,
                )
            )

    def test_higher_scope_can_replace_lower_scope(self):

        store = ParameterStore()

        store.set(
            Parameter(
                name="x",
                value="local",
                datatype="String",
                scope=ParameterScope.LOCAL,
            )
        )

        store.set(
            Parameter(
                name="x",
                value="global",
                datatype="String",
                scope=ParameterScope.GLOBAL,
            )
        )

        assert store.get("x") == "global"

class TestSerializers:

            operator="Equals",
            match_mode="Any",
            right_type="User_Value",

        )

        data = serializer.serialize(

        restored = serializer.deserialize(
            data
        )

        assert restored.id == condition.id
        assert restored.left_type == condition.left_type
        assert restored.left_name == condition.left_name
        assert restored.operator == condition.operator
        assert restored.right_type == condition.right_type
        assert restored.right_value == condition.right_value

                left_name="status",

                right_type="User_Value",
                right_value="approved",
            )
        )

        data = serializer.serialize_group(
            group
        )

        restored = serializer.deserialize_group(
            data

        version_2_configuration.rulesets[0].name = (
            "Version Two"
        )

        version_2 = service.create_version(
            version_2_configuration,
            name="Version Two"
        )


        # -------------------------------------------------
        # Switch back to Version 1
        # -------------------------------------------------

        result = service.switch_version(
            version_1.id
        )

        assert result is version_1

        assert version_1.status == "WORKING"

        assert version_2.status == "AVAILABLE"


        # -------------------------------------------------

        # -------------------------------------------------

        version_3_configuration = (
            __import__("copy").deepcopy(

        )

            == "Version Two"
        )

            == "Version Three"
        )

        versions = service.list_versions()

    def test_full_version_branching_and_rollback(self):

        repository = VersionRepository()

        service = VersionService(
            repository=repository
        )

        # --------------------------------------------------
        # Version 1 - base configuration
        # --------------------------------------------------

        configuration = FirewallConfiguration()

        configuration.add_ruleset(
            RulesetConfig(
                name="Base Security"
            )
        )

        version1 = service.create_version(
            configuration=configuration,
            name="Base Version",
            description="Initial firewall configuration"
        )

        assert version1.number == 1
        assert version1.parent_version_id is None
        assert len(
            version1.configuration.rulesets
        ) == 1

        # --------------------------------------------------
        # Version 2 - created from Version 1
        # --------------------------------------------------

        version2 = service.create_version_from_version(
            version1.id,
            name="Web Traffic Update",
            description="Allow web traffic"
        )

        assert version2.number == 2
        assert version2.parent_version_id == version1.id

        # Version 2 starts with Version 1 configuration
        assert len(
            version2.configuration.rulesets
        ) == 1

        assert (
            version2.configuration.rulesets[0].name
            == "Base Security"
        )

        # --------------------------------------------------
        # Modify Version 2
        # --------------------------------------------------

        ruleset2 = (
            version2.configuration.rulesets[0]
        )

        ruleset2.add_rule(
            RuleConfig(
                name="Allow Web Traffic",
                order=1
            )
        )

        service._save_version(
            version2
        )

        # Version 2 has the new rule
        assert len(
            version2.configuration.rulesets[0].rules
        ) == 1

        # Version 1 must remain unchanged
        assert len(
            version1.configuration.rulesets[0].rules
        ) == 0

        # --------------------------------------------------
        # Version 3 - created from Version 2
        # --------------------------------------------------

        version3 = service.create_version_from_version(
            version2.id,
            name="Version 3",
            description="Further web configuration"
        )

        assert version3.number == 3
        assert version3.parent_version_id == version2.id

        # Version 3 inherits Version 2's rule
        assert len(
            version3.configuration.rulesets[0].rules
        ) == 1

        assert (
            version3.configuration.rulesets[0].rules[0].name
            == "Allow Web Traffic"
        )

        # --------------------------------------------------
        # Roll back to Version 1
        # --------------------------------------------------

        service.switch_version(
            version1.id
        )

        assert version1.status == "WORKING"

        assert version2.status == "AVAILABLE"
        assert version3.status == "AVAILABLE"

        # --------------------------------------------------
        # Version 4 - new branch from Version 1
        # --------------------------------------------------

        version4 = service.create_version_from_version(
            version1.id,
            name="Version 4",
            description="New configuration after rollback"
        )

        assert version4.number == 4

        # Version 4 branches from Version 1
        assert version4.parent_version_id == version1.id

        # Version 4 inherits Version 1's configuration,
        # NOT Version 2 or Version 3's configuration.
        assert len(
            version4.configuration.rulesets
        ) == 1

        assert len(
            version4.configuration.rulesets[0].rules
        ) == 0

        # --------------------------------------------------
        # All four versions must still exist
        # --------------------------------------------------

        versions = service.list_versions()

        assert len(versions) == 4

        assert [version.number for version in versions] == [
            1,
            2,
            3,
            4
        ]

        # --------------------------------------------------
        # Verify parent relationships
        # --------------------------------------------------

        assert version1.parent_version_id is None

        assert version2.parent_version_id == version1.id

        assert version3.parent_version_id == version2.id

        assert version4.parent_version_id == version1.id

        # --------------------------------------------------
        # Verify old configurations remain intact
        # --------------------------------------------------

        assert len(
            version1.configuration.rulesets[0].rules
        ) == 0

        assert len(
            version2.configuration.rulesets[0].rules
        ) == 1

        assert len(
            version3.configuration.rulesets[0].rules
        ) == 1

        assert len(
            version4.configuration.rulesets[0].rules
        ) == 0
